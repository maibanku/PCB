# 引擎地基实现计划（Engine Foundation）

> 📊 **实现进度**：`[██████████] 100%` · 10/10 Task · ✅ 已实现 · 追踪：[[00-总览/06-前期搭建项目路线]] · 最后更新：2026-07-27

> **For agentic workers:** REQUIRED SUB-SKILL: 使用 `superpowers:subagent-driven-development`（推荐）或 `superpowers:executing-plans` 按任务逐个实现。步骤使用 `- [ ]` 复选框跟踪。

**Goal:** 搭建自研 EDA 引擎的地基——平台抽象 + GPU 渲染服务器抽象（双驱动 OpenGL/Vulkan）+ 输入 + 2D 画布 + 主循环，最终产出"一个能显示 EDA 风格矢量图形（焊盘矩形、走线、网格）并响应鼠标缩放平移的窗口"。

**Architecture:** 完全对齐 Godot 的目录与职责划分。`core/` 放值类型与输入（Object/ClassDB 反射体系留到 Plan 2）；`servers/` 放后端无关的服务抽象（`RenderingServer`）；`drivers/` 放各后端实现（`RasterizerGL` / `RasterizerVK`）；`platform/` 放窗口与事件循环（GLFW 引导，后续替换为自研 DisplayServer）；`scene/2d/` 放 `Canvas2D`（立即模式画布，CPU 侧把图元栅格化为三角形顶点）；`main/` 放主循环。头文件与源文件同目录（Godot 风格，不分 include/src）。

**Tech Stack:** C++20、CMake ≥ 3.20、GLFW（窗口/输入引导）、glad（OpenGL loader）、Vulkan SDK、GoogleTest、CMake FetchContent。

```cpp
#pragma once
#include <cmath>

namespace eda {

struct Vector2 {
    float x = 0.0f, y = 0.0f;
    Vector2() = default;
    Vector2(float x_, float y_) : x(x_), y(y_) {}
    Vector2 operator+(const Vector2& o) const { return {x + o.x, y + o.y}; }
    Vector2 operator-(const Vector2& o) const { return {x - o.x, y - o.y}; }
    Vector2 operator*(float s) const { return {x * s, y * s}; }
    Vector2 operator/(float s) const { return {x / s, y / s}; }
    bool operator==(const Vector2& o) const { return x == o.x && y == o.y; }
    float length() const { return std::sqrt(x * x + y * y); }
    Vector2 normalized() const {
        float l = length();
        return l > 0 ? Vector2{x / l, y / l} : Vector2{};
    }
};

struct Vector2i {
    int x = 0, y = 0;
};

}  // namespace eda
```

```cpp
#include "vector2.h"
namespace eda {

// 2D 仿射变换（对应 Godot Transform2D）：行主序 3x3，仅用 6 个有效分量
struct Transform2D {
    float m[9] = {1,0,0, 0,1,0, 0,0,1};
    static Transform2D translate(Vector2 t) {
        Transform2D r; r.m[2] = t.x; r.m[5] = t.y; return r;
    }
    static Transform2D scale(Vector2 s) {
        Transform2D r; r.m[0] = s.x; r.m[4] = s.y; return r;
    }
    Vector2 transform_point(Vector2 p) const {
        return {m[0]*p.x + m[1]*p.y + m[2], m[3]*p.x + m[4]*p.y + m[5]};
    }
};

}  // namespace eda
```

```cpp
#pragma once
#include "core/input/input_event.h"
#include <vector>
#include <unordered_set>

namespace eda {

class InputState {
public:
    void push(InputEvent e);
    bool poll(InputEvent& out);
    bool is_key_down(int key) const { return keys_down_.count(key) > 0; }
    Vector2 mouse_position() const { return mouse_pos_; }
    void clear() { events_.clear(); }
private:
    std::vector<InputEvent> events_;
    std::unordered_set<int> keys_down_;
    Vector2 mouse_pos_{};
};

}  // namespace eda
```

`core/input/input_state.cpp`：

```cpp
#include "input_state.h"
namespace eda {

void InputState::push(InputEvent e) {
    if (e.type == InputEvent::Type::KEY) {
        if (e.action == KeyAction::PRESS || e.action == KeyAction::REPEAT)
            keys_down_.insert(e.key);
        else if (e.action == KeyAction::RELEASE)
            keys_down_.erase(e.key);
    } else if (e.type == InputEvent::Type::MOUSE_MOVE) {
        mouse_pos_ = e.mouse_pos;
    }
    events_.push_back(e);
}

bool InputState::poll(InputEvent& out) {
    if (events_.empty()) return false;
    out = events_.front();
    events_.erase(events_.begin());
    return true;
}

}  // namespace eda
```

```cpp
#include "rasterizer_gl.h"
#include <glad/gl.h>
#include <GLFW/glfw3.h>

namespace eda {

// vertex: 屏幕像素坐标 → 用 uniform u_res 转 NDC（y 翻转）
static const char* kVS = R"(
#version 330 core
in vec2 a_pos; in vec4 a_color;
uniform vec2 u_res;
out vec4 v_color;
void main(){
    vec2 ndc = (a_pos / u_res) * 2.0 - 1.0;
    gl_Position = vec4(ndc.x, -ndc.y, 0.0, 1.0);
    v_color = a_color;
})";
static const char* kFS = R"(
#version 330 core
in vec4 v_color; out vec4 frag;
void main(){ frag = v_color; })";

unsigned int RasterizerGL::compile_program_() {
    auto mk = [](GLenum type, const char* src) {
        unsigned int s = glCreateShader(type);
        glShaderSource(s, 1, &src, nullptr); glCompileShader(s); return s;
    };
    unsigned int vs = mk(GL_VERTEX_SHADER, kVS);
    unsigned int fs = mk(GL_FRAGMENT_SHADER, kFS);
    unsigned int p = glCreateProgram();
    glAttachShader(p, vs); glAttachShader(p, fs); glLinkProgram(p);
    glDeleteShader(vs); glDeleteShader(fs);
    return p;
}

void RasterizerGL::initialize(void* /*window*/) {
    gladLoadGL(glfwGetProcAddress);
    shader_ = compile_program_();
    loc_resolution_ = glGetUniformLocation(shader_, "u_res");
    glGenVertexArrays(1, &vao_); glBindVertexArray(vao_);
    glGenBuffers(1, &vbo_);
    initialized_ = true;
}
void RasterizerGL::shutdown() {
    if (!initialized_) return;
    glDeleteBuffers(1, &vbo_); glDeleteVertexArrays(1, &vao_); glDeleteProgram(shader_);
    initialized_ = false;
}
void RasterizerGL::begin_frame() {}
void RasterizerGL::end_frame() { glFlush(); }
void RasterizerGL::clear(Color c) {
    glClearColor(c.r, c.g, c.b, c.a); glClear(GL_COLOR_BUFFER_BIT);
}
void RasterizerGL::set_viewport(int, int, int w, int h) {
    viewport_w_ = w; viewport_h_ = h; glViewport(0, 0, w, h);
}
void RasterizerGL::draw_triangles(const std::vector<CanvasVertex>& vertices) {
    if (vertices.empty()) return;
    glUseProgram(shader_);
    glUniform2f(loc_resolution_, (float)viewport_w_, (float)viewport_h_);
    glBindVertexArray(vao_);
    glBindBuffer(GL_ARRAY_BUFFER, vbo_);
    glBufferData(GL_ARRAY_BUFFER, vertices.size() * sizeof(CanvasVertex), vertices.data(), GL_DYNAMIC_DRAW);
    glEnableVertexAttribArray(0);
    glVertexAttribPointer(0, 2, GL_FLOAT, GL_FALSE, sizeof(CanvasVertex), (void*)0);
    glEnableVertexAttribArray(1);
    glVertexAttribPointer(1, 4, GL_FLOAT, GL_FALSE, sizeof(CanvasVertex), (void*)(2 * sizeof(float)));
    glDrawArrays(GL_TRIANGLES, 0, (int)vertices.size());
}

}  // namespace eda
```

> 说明：`CanvasVertex` 布局为 `Vector2(8) + Color(16) = 24` 字节；shader 用 uniform `u_res` 把像素坐标转 NDC，与 `Canvas2D` 的屏幕空间输出对齐。

- [ ] **Step 3: 按标准序列实现各私有方法**

`drivers/vulkan/rasterizer_vk.cpp` 按以下顺序实现（每实现一个方法即编译验证；`#define VK_CHECK(x) do{VkResult r=x; if(r!=VK_SUCCESS) throw std::runtime_error("vk");}while(0)`）：

1. `create_instance_()`：`vkCreateInstance`（`VkApplicationInfo` 填引擎名/版本；Debug 加 `VK_LAYER_KHRONOS_validation`）
2. `create_surface_(window)`：`glfwCreateWindowSurface(instance_, (GLFWwindow*)window, nullptr, &surface_)`
3. `pick_physical_device_()`：`vkEnumeratePhysicalDevices`，选首个支持图形队列族（`VK_QUEUE_GRAPHICS_BIT`）且 `glfwGetPhysicalDevicePresentationSupport` 为真的设备与队列族 `queue_family_`
4. `create_logical_device_()`：`vkCreateDevice`，队列优先级 1.0，启用 `VK_KHR_swapchain`
5. `create_swapchain_()`：查询 `vkGetPhysicalDeviceSurfaceCapabilitiesKHR`，选 `VK_FORMAT_B8G8R8A8_UNORM` + `VK_COLOR_SPACE_SRGB_NONLINEAR_KHR`，`imageCount = minImageCount+1`，`presentMode = MAILBOX`（不支持则 FIFO），`imageExtent = {width_, height_}`
6. `create_render_pass_()`：单 color attachment，`loadOp = CLEAR`、`storeOp = STORE`、`initialLayout = UNDEFINED`、`finalLayout = PRESENT_SRC`
7. `create_pipeline_()`：顶点 shader 接收 `pos(vec2)+color(vec4)`（与 GL 同语义，可用 glslang/SPIR-V 或 shaderc 预编译为 SPIR-V 字节码内嵌）；`VkVertexInputBindingDescription stride=24`，`location 0: R32G32_SFLOAT`、`location 1: R32G32B32A32_SFLOAT`；拓扑 `TRIANGLE_LIST`；视口/scissor 动态
8. `create_framebuffers_()`：为每个 swapchain image 创建 `VkImageView` + `VkFramebuffer`（尺寸 `{width_, height_}`)
9. `create_command_pool_and_buffers_()`：`vkCreateCommandPool` + `vkAllocateCommandBuffers`（每 image 一个）
10. `create_sync_()`：两个 `VkSemaphore` + 一个 `VkFence`
11. `create_vbo_()`：`vkCreateBuffer` + `vkAllocateMemory`（`HOST_VISIBLE_BIT | HOST_COHERENT_BIT`）+ `vkBindBufferMemory`

> 关键 API 序列核对清单（实现时逐项）：`vkCreateInstance` → `glfwCreateWindowSurface` → `vkEnumeratePhysicalDevices`/`vkGetPhysicalDeviceQueueFamilyProperties`/`glfwGetPhysicalDevicePresentationSupport` → `vkCreateDevice`/`vkGetDeviceQueue` → `vkCreateSwapchainKHR`/`vkGetSwapchainImagesKHR` → `vkCreateImageView`×N → `vkCreateRenderPass` → `vkCreateShaderModule`×2 → `vkCreateGraphicsPipelines` → `vkCreateFramebuffer`×N → `vkCreateCommandPool`/`vkAllocateCommandBuffers` → `vkCreateSemaphore`×2/`vkCreateFence` → `vkCreateBuffer`/`vkAllocateMemory`。

- [ ] **Step 4: 实现帧绘制**

```cpp
void RasterizerVK::begin_frame() {
    VK_CHECK(vkWaitForFences(device_, 1, &in_flight_, VK_TRUE, UINT64_MAX));
    VK_CHECK(vkResetFences(device_, 1, &in_flight_));
    uint32_t idx;
    VK_CHECK(vkAcquireNextImageKHR(device_, swapchain_, UINT64_MAX, img_available_, VK_NULL_HANDLE, &idx));
    // 录制 cmd_buffers_[idx]：begin → begin_render_pass(clear_color_) → bind pipeline →
    //   set viewport/scissor({0,0,width_,height_}) → bind vbo → draw(tri_count_*3) →
    //   end_render_pass → end
}
void RasterizerVK::end_frame() {
    VkSubmitInfo si{VK_STRUCTURE_TYPE_SUBMIT_INFO};
    // waitSemaphore=img_available_, signalSemaphore=render_done_, cmdBuffers=&cmd_buffers_[idx]
    VK_CHECK(vkQueueSubmit(queue_, 1, &si, in_flight_));
    VkPresentInfoKHR pi{VK_STRUCTURE_TYPE_PRESENT_INFO_KHR};
    // waitSemaphore=render_done_, swapchain, imageIndices=&idx
    vkQueuePresentKHR(queue_, &pi);
}
void RasterizerVK::clear(Color c) { clear_color_ = c; }
void RasterizerVK::set_viewport(int, int, int w, int h) { width_ = w; height_ = h; }
void RasterizerVK::draw_triangles(const std::vector<CanvasVertex>& v) {
    if (v.empty()) { tri_count_ = 0; return; }
    void* mapped = nullptr;
    vkMapMemory(device_, vbo_mem_, 0, v.size() * sizeof(CanvasVertex), 0, &mapped);
    std::memcpy(mapped, v.data(), v.size() * sizeof(CanvasVertex));
    vkUnmapMemory(device_, vbo_mem_);
    tri_count_ = (uint32_t)v.size() / 3;
}
```

```cpp
    vkDestroyPipeline(device_, pipeline_, nullptr);
    vkDestroyPipelineLayout(device_, pipeline_layout_, nullptr);
    vkDestroyRenderPass(device_, render_pass_, nullptr);
    for (auto& v : views_) vkDestroyImageView(device_, v, nullptr);
    vkDestroySwapchainKHR(device_, swapchain_, nullptr);
    vkDestroyDevice(device_, nullptr);
    vkDestroySurfaceKHR(instance_, surface_, nullptr);
    vkDestroyInstance(instance_, nullptr);
}
```

```cpp
#include "canvas_2d.h"
#include "servers/rendering_server.h"
#include <cmath>

namespace eda {

Canvas2D::Canvas2D(RenderingServer* server) : server_(server) {}
void Canvas2D::resize(int w, int h) { width_ = w; height_ = h; }
void Canvas2D::set_camera(Vector2 offset, float zoom) { cam_offset_ = offset; cam_zoom_ = zoom; }

Vector2 Canvas2D::world_to_screen(Vector2 p) const {
    return { (p.x - cam_offset_.x) * cam_zoom_, (p.y - cam_offset_.y) * cam_zoom_ };
}
void Canvas2D::begin(Color clear_color) {
    vertices_.clear();
    if (server_) { server_->set_viewport(0, 0, width_, height_); server_->clear(clear_color); }
}
void Canvas2D::end() {
    if (server_) server_->draw_triangles(vertices_);
    vertices_.clear();
}
void Canvas2D::push_triangle(Vector2 a, Vector2 b, Vector2 c, Color color) {
    vertices_.push_back({world_to_screen(a), color});
    vertices_.push_back({world_to_screen(b), color});
    vertices_.push_back({world_to_screen(c), color});
}
void Canvas2D::push_line_as_quad(Vector2 a, Vector2 b, float thickness, Color color) {
    Vector2 d = (b - a).normalized();
    Vector2 n{d.y, -d.x};
    Vector2 h = n * (thickness * 0.5f);
    Vector2 p0 = a - h, p1 = a + h, p2 = b + h, p3 = b - h;
    push_triangle(p0, p1, p2, color);
    push_triangle(p0, p2, p3, color);
}
void Canvas2D::draw_line(Vector2 a, Vector2 b, Color color, float thickness) {
    push_line_as_quad(a, b, thickness, color);
}
void Canvas2D::draw_rect(Vector2 pos, Vector2 size, Color color, bool filled) {
    Vector2 a = pos, b{pos.x + size.x, pos.y}, c{pos.x + size.x, pos.y + size.y}, d{pos.x, pos.y + size.y};
    if (filled) {
        push_triangle(a, b, c, color);
        push_triangle(a, c, d, color);
    } else {
        draw_line(a, b, color, 1.0f); draw_line(b, c, color, 1.0f);
        draw_line(c, d, color, 1.0f); draw_line(d, a, color, 1.0f);
    }
}
void Canvas2D::draw_polygon(const std::vector<Vector2>& pts, Color color, bool filled) {
    if (pts.size() < 3) return;
    if (filled) {
        for (size_t i = 1; i + 1 < pts.size(); ++i)
            push_triangle(pts[0], pts[i], pts[i + 1], color);
    } else {
        for (size_t i = 0; i < pts.size(); ++i)
            draw_line(pts[i], pts[(i + 1) % pts.size()], color, 1.0f);
    }
}
void Canvas2D::draw_circle(Vector2 center, float radius, Color color, bool filled, int segments) {
    if (segments < 3) segments = 3;
    constexpr float TAU = 6.2831853f;
    if (filled) {
        for (int i = 0; i < segments; ++i) {
            float a0 = (float)i / segments * TAU, a1 = (float)(i + 1) / segments * TAU;
            Vector2 p0{center.x + radius * std::cos(a0), center.y + radius * std::sin(a0)};
            Vector2 p1{center.x + radius * std::cos(a1), center.y + radius * std::sin(a1)};
            push_triangle(center, p0, p1, color);
        }
    } else {
        std::vector<Vector2> pts(segments);
        for (int i = 0; i < segments; ++i) {
            float a = (float)i / segments * TAU;
            pts[i] = {center.x + radius * std::cos(a), center.y + radius * std::sin(a)};
        }
        draw_polygon(pts, color, false);
    }
}

}  // namespace eda
```

```cpp
#include "platform/window.h"
#include "platform/glfw/glfw_window.h"
#include "drivers/opengl/rasterizer_gl.h"  // 默认 GL 驱动
#include "scene/2d/canvas_2d.h"
#include <algorithm>

int main() {
    constexpr int W = 1280, H = 800;
    auto window = eda::create_glfw_window(W, H, "EDA Engine - Foundation", eda::RenderingDriver::OPENGL);
    eda::RasterizerGL server;
    server.initialize(window->native_handle());

    eda::Canvas2D canvas(&server);
    canvas.resize(W, H);

    eda::Vector2 cam_offset{0, 0};
    float cam_zoom = 1.0f;

    while (!window->should_close()) {
        window->poll_events();
        auto* glfw_win = static_cast<eda::GlfwWindow*>(window.get());
        auto& input = glfw_win->input_state();

        eda::InputEvent e;
        while (input.poll(e)) {
            if (e.type == eda::InputEvent::Type::MOUSE_SCROLL) {
                cam_zoom *= (e.scroll_delta > 0) ? 1.1f : 0.9f;
                cam_zoom = std::clamp(cam_zoom, 0.05f, 50.0f);
            }
        }
        // 中键拖拽平移：记录按下位置差实现（留作扩展）

        canvas.set_camera(cam_offset, cam_zoom);
        canvas.begin({0.12f, 0.12f, 0.13f, 1.0f});

        eda::Color grid{0.2f, 0.2f, 0.22f, 1.0f};
        constexpr float G = 10.0f;
        for (float x = -500; x <= 500; x += G) canvas.draw_line({x, -500}, {x, 500}, grid, 0.5f);
        for (float y = -500; y <= 500; y += G) canvas.draw_line({-500, y}, {500, y}, grid, 0.5f);

        canvas.draw_rect({20, 20}, {4, 4}, {0.9f, 0.8f, 0.2f, 1.0f}, true);
        canvas.draw_rect({40, 20}, {4, 4}, {0.9f, 0.8f, 0.2f, 1.0f}, true);
        canvas.draw_line({22, 22}, {42, 22}, {0.3f, 0.7f, 1.0f, 1.0f}, 1.0f);
        canvas.draw_circle({60, 22}, 2, {0.9f, 0.4f, 0.4f, 1.0f}, true, 12);

        canvas.end();
        window->swap_buffers();
    }
    server.shutdown();
    return 0;
}
```
